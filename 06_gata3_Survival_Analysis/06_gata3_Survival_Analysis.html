t.test.all.genes <- function(x, s1, s2) {
x1 <- as.numeric(x[s1])
x2 <- as.numeric(x[s2])
t.out <- t.test(x1, x2, alternative = "two.sided", var.equal = TRUE)
return(t.out$p.value)
}

rawp_gender <- apply(brain_aging[g.g, ], 1, t.test.all.genes, s1 = colnames(brain_aging_female), s2 = colnames(brain_aging_male))
rawp_age <- apply(brain_aging[g.a, ], 1, t.test.all.genes, s1 = colnames(brain_age_less50), s2 = colnames(brain_age_more50))

# Adjust p-values
gender_p_adj <- p.adjust(rawp_gender, method = "holm")
age_p_adj <- p.adjust(rawp_age, method = "holm")

## 3. P-Value Visualization

## 4. TCGA Breast Cancer RNA-seq Analysis

# Load Data
tcga <- read.table("/Users/gauri1399/Desktop/Classes/Sem3/Gene expression/Lab6/tcga_brca_fpkm.txt", header=TRUE, row.names=1, check.names=FALSE)
tcga_annotation <- read.table("/Users/gauri1399/Desktop/Classes/Sem3/Gene expression/Lab6/tcga_brca_fpkm_sam.txt", header=TRUE, sep="\t", stringsAsFactors=FALSE, check.names=FALSE)

# GATA3 Expression Grouping
gata3_idx <- grep("^GATA3(\\b|\\|)", rownames(tcga), perl = TRUE)
gata3 <- as.numeric(tcga[gata3_idx[1], ])
threshold <- quantile(gata3, 0.75, na.rm = TRUE)
gata3_expression_group <- ifelse(gata3 >= threshold, 1, 0)

## 5. Survival Analysis

time <- as.numeric(tcga_annotation$months_to_event)
status <- as.integer(toupper(tcga_annotation$vital_status) == "LIVING")
keep <- complete.cases(time, status, gata3_expression_group)
data <- data.frame(
time = time[keep],
status = status[keep],
group = gata3_expression_group[keep]
)

# Kaplan-Meier
km_fit <- survfit(Surv(time, status) ~ group, data=data)
km_test <- survdiff(Surv(time, status) ~ group, data=data)
km.p <- 1 - pchisq(km_test$chisq, df = length(km_test$n)-1)

# Cox PH
coxfit <- coxph(Surv(time, status) ~ group, data=data)
summ_coxfit <- summary(coxfit)
hazard_ratio <- unname(summ_coxfit$coef["group", "exp(coef)"])
p_cox <- summ_coxfit$coef["group", "Pr(>|z|)"]

# Plot KM
plot(km_fit, lwd=2, col=c("blue","orange"), xlab="Time (months)", ylab="Survival probability",
main="GATA3: Upper Quartile vs Others")
legend("topright", legend=c(paste0("Others (n=", table(data$group==0), ")"),
paste0("Upper 25% (n=", table(data$group==1), ")")),
col=c("blue","orange"), lwd=2, bty="n")
mtext(sprintf("KM p=%.3g | Cox p=%.3g | HR=%.2f", km.p, p_cox, hazard_ratio),
side=3, line=0.5, cex=0.9)
